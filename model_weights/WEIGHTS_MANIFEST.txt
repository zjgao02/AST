# ASTevolve model and asset placement manifest
#
# This file lists non-Git assets needed on a Linux/GPU machine.
# Download or copy each asset into the target path relative to project root.
# Large files under model_weights/ are intentionally ignored by Git.

[required] ProGen2 small sequence prior
target_dir: model_weights/progen2-small
source: Hugging Face model repository hugohrban/progen2-small
local_windows_source_if_available: D:\Downloads\progen2-small
required_files:
  config.json
  configuration_progen.py
  generation_config.json
  modeling_progen.py
  tokenizer.json
  model.safetensors
env:
  ASTEVOLVE_PROGEN_MODEL_DIR=${ASTEVOLVE_MODEL_ROOT}/progen2-small
notes:
  ASTevolve loads this model with local_files_only=True.
  The target directory must contain the files directly, not only a Hugging Face cache snapshot.

[required] Protenix package/runtime weights
target_dir: model_weights/protenix
source: Protenix release/package documentation for the installed protenix version
model_name_used_by_ast: protenix_mini_esm_v0.5.0
env:
  ASTEVOLVE_PROTENIX_ROOT=${ASTEVOLVE_MODEL_ROOT}/protenix
  PROTENIX_ROOT_DIR=${ASTEVOLVE_PROTENIX_ROOT}
notes:
  The Python package is installed through environments/pytorch-linux.yml or requirements-gpu.txt.
  Keep any downloaded Protenix checkpoints/cache under this directory on the Linux machine.
  The exact checkpoint filenames are Protenix-version-specific; verify with:
    protenix predict --help
    python -c "import protenix, pathlib; print(pathlib.Path(protenix.__file__).parent)"
  For protein-protein or peptide-complex cases, protenix_mini_esm_v0.5.0 can
  OOM during ESM3B embedding precomputation on some A100 jobs. Download at
  least one non-ESM fallback checkpoint and run with
  ASTEVOLVE_PROTENIX_MODEL_NAME=protenix_mini_default_v0.5.0 or
  ASTEVOLVE_PROTENIX_MODEL_NAME=protenix_tiny_default_v0.5.0.
optional_fallback_files:
  release_data/checkpoint/protenix_mini_default_v0.5.0.pt
    url: https://af3-dev.tos-cn-beijing.volces.com/release_model/protenix_mini_default_v0.5.0.pt
  release_data/checkpoint/protenix_tiny_default_v0.5.0.pt
    url: https://af3-dev.tos-cn-beijing.volces.com/release_model/protenix_tiny_default_v0.5.0.pt
